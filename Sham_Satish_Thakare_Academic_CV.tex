\section*{Theoretical \& Methodological Contributions}
\textbf{Credit Assignment in Reasoning Reinforcement Learning (RLVR)} \hfill \textcolor{accent}{\textbf{Mathematical ML}}
\begin{itemize}
    \item Analyzed token predictive entropy $H(Y_t \mid Y_{<t}, X)$ across mathematical reasoning derivations on GSM8K and MATH.
    \item Proved that predictive entropy correlates strongly with reasoning sequence length ($r = +0.486$, $p < 10^{-6}$), acting as an implicit length penalty that systematically penalizes valid multi-step derivations in uncertainty-weighted GRPO.
    \item Conducted five-way matched ablation study proving that consistency-weighted GRPO yields $0.00\%$ Pass@1 gain over length-controlled baselines, establishing empirical bounds on proxy reliability.
\end{itemize}

\vspace{2pt}

\textbf{Failure-Domain Aware Consensus Quorum Adaptation} \hfill \textcolor{accent}{\textbf{Distributed Systems Theory}}
\begin{itemize}
    \item Formulated online latency profiling algorithms in C++20 Raft consensus clusters that dynamically re-weight quorums.
    \item Proved that under bounded clock skew $\Delta$ and asynchronous transitions, the dynamically adapted quorums preserve non-empty intersection invariants $\mathcal{Q}_i \cap \mathcal{Q}_j \neq \emptyset$, guaranteeing linearizability across arbitrary network partitions.
    \item Implemented open-source C++20 library \texttt{quorumshift} with reproducible fault-injection test harnesses and lock-free execution queues.
\end{itemize}

\vspace{2pt}

\textbf{State-Contrast Continuation Evaluation} \hfill \textcolor{accent}{\textbf{Evaluation Methodology}}
\begin{itemize}
    \item Formulated state-contrast continuation evaluation metric ($D_{\text{recovery}} = \mathbb{E}[\Delta_{\text{recovery}}] - \mathbb{E}[\Delta_{\text{control}}]$) in \texttt{recovery\_eval}, establishing counterfactual controls that isolate true self-correction from generative capacity under frozen structural covariates.
\end{itemize}
